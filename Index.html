async function fetchMarketData() { // Ejemplo de llamada a endpoint de datos financieros o API propia try { const response = await fetch('https://query1.finance.yahoo.com/v8/finance/chart/QQQ?interval=5m&range=1d'); const data = await response.json(); const quotes = data.chart.result[0].indicators.quote[0]; const highs = quotes.high; const lows = quotes.low; const maxHigh = Math.max(...highs.filter(h => h !== null)); const minLow = Math.min(...lows.filter(l => l !== null)); console.log(`Máximo de sesión: ${maxHigh}`); console.log(`Mínimo de sesión: ${minLow}`); } catch (error) { console.error("Error al obtener datos de mercado:", error); } } fetchMarketData();
